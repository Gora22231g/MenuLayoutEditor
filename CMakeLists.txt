cmake_minimum_required(VERSION 3.25)
project(MenuLayoutEditor VERSION 1.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Set GEODE_SDK to your Geode SDK directory")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/FileDialog.cpp)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
set_target_properties(${PROJECT_NAME} PROPERTIES DISABLE_PRECOMPILE_HEADERS ON)
option(LAYOUT_BUILD_TESTS "Build geometry regression tests" OFF)
if(LAYOUT_BUILD_TESTS)
  enable_testing()
  add_executable(LayoutGeometryTests tests/geometry.cpp)
  add_test(NAME layout-geometry COMMAND LayoutGeometryTests)
  add_executable(LayoutConfigTests tests/config.cpp)
  target_link_libraries(LayoutConfigTests PRIVATE mat-json-impl)
  add_test(NAME layout-config COMMAND LayoutConfigTests)
  if(WIN32)
    add_executable(LayoutGuideShaderTests tests/guide_shader.cpp)
    target_link_libraries(LayoutGuideShaderTests PRIVATE opengl32 user32 gdi32)
    add_test(NAME layout-guide-shader COMMAND LayoutGuideShaderTests)
    set_tests_properties(layout-guide-shader PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
